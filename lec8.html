// 1. الحصول على العناصر من الصفحة
const btn = document.getElementById('btn');
const title = document.getElementById('title');

// 2. إضافة مستمع حدث (Event Listener) للزر

  // 3. تغيير نص العنوان عند النقر

  // 4. تعطيل الزر بعد النقر
  btn.disabled = true;
  
  // 5. تغيير نص الزر
  btn.innerText = 'تم النقر بالفعل';
});

مقدمة إلى Bootstrap (الإصدار الحديث 5.x)

ما هو Bootstrap؟

إطار عمل CSS (مع أجزاء من JS) يجعل بناء واجهات متجاوبة سريعًا عبر نظام شبكة (Grid) ومكونات جاهزة. طورته Twitter وأصبح معيارًا للتصميم السريع والمتجاور لمواقع الويب.

كيف تضيف Bootstrap لصفحتك؟

أبسط طريقة عبر CDN (في head):

<!-- CSS -->
<link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/css/bootstrap.min.css" rel="stylesheet">

<!-- JS -->
<script src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/js/bootstrap.bundle.min.js" defer></script>

في بيئة تعليمية، يمكن استخدام النسخ المحلية أيضاً عن طريق التحميل من الموقع الرسمي.

قلم تصميم احترافي

قلم احترافي للرسم الرقمي مع دقة عالية وحساسية للضغط.

تفاصيل المنتج

هذا القلم مصنوع من مواد عالية الجودة ويعمل مع الأجهزة اللوحية وأجهزة الكمبيوتر.

  • دقة: 8192 مستوى للضغط
  • متوافق مع: Windows, macOS, Android
  • يأتي مع 10 رؤوس قابلة للاستبدال
  • ضمان: 3 سنوات

كود JavaScript المستخدم

// الانتظار حتى يتم تحميل DOM بالكامل
document.addEventListener('DOMContentLoaded', () => {
  // الحصول على العناصر
  const addBtn = document.getElementById('addBtn');
  const cartCount = document.getElementById('cartCount');
  const buyNow = document.getElementById('buyNow');
  
  // حدث النقر على زر "أضف إلى السلة"
  addBtn.addEventListener('click', () => {
    // زيادة عداد السلة بمقدار 1
    cartCount.innerText = parseInt(cartCount.innerText) + 1;
    
    // تغيير مؤقت للزر للإشارة إلى النجاح
    const originalText = addBtn.innerText;
    addBtn.innerText = 'تمت الإضافة!';
    addBtn.classList.remove('btn-primary');
    addBtn.classList.add('btn-success');
    
    // العودة إلى الحالة الأصلية بعد ثانيتين
    setTimeout(() => {
      addBtn.innerText = originalText;
      addBtn.classList.remove('btn-success');
      addBtn.classList.add('btn-primary');
    }, 2000);
  });
  
  // حدث النقر على زر "شراء الآن" في المودال
  buyNow.addEventListener('click', () => {
    // عرض رسالة تأكيد
    alert('شكرًا! تم البدء بعملية الشراء (توضيحي).');
    
    // إغلاق المودال برمجياً
    const modal = bootstrap.Modal.getInstance(document.getElementById('detailsModal'));
    modal.hide();
  });
});

عناصر JavaScript + Bootstrap التي يجب إيضاحها للطلاب عمليًا

1. DOM Traversal & Manipulation

الوصول إلى العناصر

// الحصول على عنصر بواسطة id
const element = document.getElementById('myId');

// الحصول على العنصر الأول بمحدد CSS
const firstElement = document.querySelector('.myClass');

// الحصول على جميع العناصر بمحدد CSS
const allElements = document.querySelectorAll('.myClass');

تعديل العناصر

// تغيير النص
element.textContent = 'نص جديد';

// تغيير HTML
element.innerHTML = '<strong>نص عريض</strong>';

// إضافة/إزالة/تبديل CSS classes
element.classList.add('active');
element.classList.remove('hidden');
element.classList.toggle('visible');

2. التعامل مع الأحداث (Events)

ربط الأحداث

// حدث النقر
button.addEventListener('click', (event) => {
  console.log('تم النقر!');
});

// حدث إرسال نموذج
form.addEventListener('submit', (event) => {
  event.preventDefault(); // منع الإرسال الافتراضي
  // التحقق من النموذج هنا
});

// حدث إدخال نص
input.addEventListener('input', (event) => {
  console.log('القيمة الجديدة:', event.target.value);
});

أحداث شائعة

  • click - نقر الماوس
  • submit - إرسال نموذج
  • input - تغيير قيمة حقل إدخال
  • change - تغيير قيمة عنصر (select, checkbox)
  • load - انتهاء تحميل الصفحة أو صورة

3. استخدام مكونات Bootstrap عبر JS

التحكم في المودال برمجيًا

// إنشاء مثيل للمودال
const myModal = new bootstrap.Modal(document.getElementById('myModal'));

// إظهار المودال
myModal.show();

// إخفاء المودال
myModal.hide();

// التبديل (إظهار/إخفاء)
myModal.toggle();

مكونات Bootstrap الأخرى

  • Collapse - للعناصر القابلة للطي
  • Toast - للإشعارات العابرة
  • Tooltip - للتلميحات (تحتاج تهيئة JS)
  • Carousel - لعرض الشرائح الدوارة
  • Dropdown - للقوائم المنسدلة

4. التحقق من النماذج (Form Validation)

التحقق باستخدام JavaScript

form.addEventListener('submit', (event) => {
  event.preventDefault(); // منع الإرسال
  
  const email = document.getElementById('email').value;
  const password = document.getElementById('password').value;
  
  // التحقق من البريد الإلكتروني
  if (!email.includes('@')) {
    showError('البريد الإلكتروني غير صالح');
    return;
  }
  
  // التحقق من كلمة المرور
  if (password.length < 6) {
    showError('كلمة المرور يجب أن تكون 6 أحرف على الأقل');
    return;
  }
  
  // إذا كان كل شيء صحيحاً
  form.submit(); // إرسال النموذج
});

function showError(message) {
  // عرض رسالة الخطأ في الـ DOM
  const errorDiv = document.getElementById('error');
  errorDiv.textContent = message;
  errorDiv.classList.remove('d-none');
}

التحقق باستخدام Bootstrap

يوفر Bootstrap فئات CSS مبنية للتحقق من النماذج:

<div class="mb-3">
  <label for="email" class="form-label">البريد الإلكتروني</label>
  <input type="email" class="form-control is-invalid" id="email">
  <div class="invalid-feedback">
    يرجى إدخال بريد إلكتروني صالح.
  </div>
</div>

المنطق الأساسي (JavaScript)

// مصفوفة لتخزين المهام
let tasks = [];

// إضافة مهمة جديدة
function addTask(taskText) {
  const task = {
    id: Date.now(), // معرف فريد
    text: taskText,
    completed: false,
    date: new Date().toLocaleDateString('ar-SA')
  };
  
  tasks.push(task);
  renderTasks();
  saveTasksToLocalStorage();
}

// عرض المهام
function renderTasks() {
  const tasksList = document.getElementById('tasksList');
  const taskCount = document.getElementById('taskCount');
  
  if (tasks.length === 0) {
    tasksList.innerHTML = '<div class="alert alert-info">لا توجد مهام</div>';
    taskCount.textContent = '0';
    return;
  }
  
  let html = '';
  tasks.forEach(task => {
    html += `
      <div class="task-item d-flex justify-content-between 
                   align-items-center p-2 mb-2 border rounded">
        <div class="form-check">
          <input type="checkbox" class="form-check-input" 
                 ${task.completed ? 'checked' : ''}
                 onchange="toggleTask(${task.id})">
          <label class="form-check-label ${task.completed ? 'text-decoration-line-through text-muted' : ''}">
            ${task.text}
          </label>
        </div>
        <button class="btn btn-sm btn-outline-danger" 
                onclick="deleteTask(${task.id})">
          <i class="fas fa-times"></i>
        </button>
      </div>
    `;
  });
  
  tasksList.innerHTML = html;
  taskCount.textContent = tasks.length;
}

مخاطر وسلبيات (تنبيهات)

مخاطر JavaScript

أداء الصفحة

كود JavaScript غير المحسّن قد يبطئ تحميل الصفحة وتفاعلها. الاستخدام المفرط للمكتبات قد يزيد وقت التحميل.

الأمان

JavaScript يعمل على جهاز المستخدم، مما يجعل التحقق من المدخلات على جانب العميل غير كافٍ. يجب دائماً التحقق على الخادم أيضاً.

إمكانية الوصول

الاعتماد الكلي على JavaScript قد يجعل الموقع غير قابل للاستخدام لمن لديهم إعاقات أو يستخدمون تقنيات مساعدة قديمة.

سلبيات Bootstrap

التشابه

الاعتماد الكلي على Bootstrap قد يجعل المواقع متشابهة. يجب تعلم CSS الأساسي للتميز وإنشاء تصاميم فريدة.

حجم الملفات

ملفات Bootstrap الكبيرة قد تزيد وقت تحميل الصفحة. يمكن استخدام أدوات مثل PurgeCSS لإزالة الأجزاء غير المستخدمة.

الاعتماد المفرط

قد يعيق Bootstrap تعلم CSS الأساسي. المطورون المبتدئون قد يعتمدون عليه دون فهم كيفية عمل CSS فعلياً.

نصائح وتوصيات

لتحسين أداء JavaScript
  • استخدم defer أو async لتحميل ملفات JS
  • قلل من المكتبات الخارجية
  • حسن الكود وتجنب الحلقات غير الضرورية
  • استخدم التخزين المؤقت (Caching)
لتحسين استخدام Bootstrap
  • تعلم CSS الأساسي أولاً
  • خصص Bootstrap بدلاً من استخدامه كما هو
  • استخدم الإصدار المخصص (Custom Build)
  • اجمع بين Bootstrap و CSS مكتوب يدوياً

التوازن مفتاح النجاح: استخدم JavaScript و Bootstrap كأدوات مساعدة، لا كبديل كامل للمعرفة الأساسية

ما دور JavaScript في صفحات الويب؟

JavaScript تضيف التفاعل الديناميكي لصفحات الويب. تتحكم في سلوك الصفحة، تتعامل مع أحداث المستخدم، تغير المحتوى ديناميكيًا، وتتحقق من النماذج.

كيف تضيف Bootstrap إلى صفحة HTML؟

الإجابة النموذجية:

عن طريق إضافة روابط CDN في head:

2. رابط JS: https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/js/bootstrap.bundle.min.js

اذكر فائدتين لدمج JavaScript مع Bootstrap.

الإجابة النموذجية:

2. تجربة مستخدم محسنة: واجهات متجاوبة وجذابة مع تفاعل ديناميكي.

لغة برمجة عالية المستوى تُنفّذ في متصفح الويب. تضيف التفاعل الديناميكي لصفحات HTML الثابتة. يمكن استخدامها أيضاً في جانب الخادم (Node.js).

BS

Bootstrap

إطار عمل CSS

إطار عمل CSS مفتوح المصدر يحتوي على قوالب تصميم جاهزة لأنماط الطباعة والنماذج والأزرار والجداول والتنقل والمكونات الأخرى.

DOM

نموذج كائن المستند

تمثيل برمجي لهيكل صفحة HTML. تسمح JavaScript بالوصول إلى عناصر DOM وتعديلها ديناميكيًا.

الأحداث

Events

أفعال أو وقائع تحدث في المتصفح (نقر، تحميل صفحة، إدخال نص). يمكن لـ JavaScript الاستماع لهذه الأحداث والاستجابة لها.

نظام الشبكة

Grid System

نظام في Bootstrap يقسم الصفحة إلى 12 عمودًا. يسمح بإنشاء تخطيطات متجاوبة تتكيف مع أحجام الشاشات المختلفة.

المودال

Modal

نافذة منبثقة تظهر فوق المحتوى الرئيسي. تستخدم لعرض محتوى إضافي دون الانتقال إلى صفحة جديدة. مكون جاهز في Bootstrap.
